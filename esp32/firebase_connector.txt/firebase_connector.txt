  Serial.println("\n4. Waiting for authentication...");

void testGetBottles() {
  Serial.println("\n=== TEST: Get All Bottles ===");
  
  String babyId = "239664840";
  std::vector<Bottle> bottles;
  
  if (firebaseService->getBottles(babyId, bottles)) {
    Serial.println("\n✓ SUCCESS! Retrieved " + String(bottles.size()) + " bottles");
    
    if (bottles.size() == 0) {
      Serial.println("No bottles found for this baby");
    } else {
      for (int i = 0; i < bottles.size(); i++) {
        Serial.println("\n--- Bottle #" + String(i + 1) + " ---");
        bottles[i].print();
      }
    }
  } else {
    Serial.println("\n✗ Failed to get bottles");
  }
}
